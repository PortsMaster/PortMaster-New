# Hollow Knight setup options, read once by the first-launch setup.
# PortMaster's setup screen asks the questions below itself; this file answers them when it
# does not (older PortMaster versions).
#
# Texture quality? Both turn the textures into ASTC, the format the handheld's GPU reads (the same size and memory). High is what this port is tested with. Low makes the setup much shorter, with slightly less exact textures.
#   medium  High (recommended)
#   fast  Low (shorter setup)
texture_quality=medium
#
# Picture shape? Hollow Knight is 16:9. (Change it later in ports/hollowknight/aspect.txt: fill or bars.)
#   fill  Fill the screen: the same width, a little more above and below (recommended)
#   bars  16:9 with black bars above and below
aspect=fill
#
# Which letters are on your handheld's buttons? (Change it later in ports/hollowknight/buttons.txt.)
#   nintendo  Nintendo: A on the right, B at the bottom (recommended)
#   xbox  Xbox: A at the bottom, B on the right
buttons=nintendo
#
# keep_installers: 1 keeps the archives or installers you put in gamedata/ after the setup;
#   by default the setup deletes them once the game is built (they are not needed to play)
keep_installers=0
